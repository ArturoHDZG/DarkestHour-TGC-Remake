	style = 1

		control = { province = 163 data = -1 }
		OR = {
			AND = {
				OR = {
					country = U25

			}
			AND = {
				OR = {
					country = U24
					country = DFR
				}
				NOT = { exists = U25 }
				NOT = { exists = DDR }
			}
		}
		flag = divided_berlin
	}
	date = { day = 1 month = january year = 1944 }
	offset = 14
	deathdate = { day = 29 month = december year = 1963 }
	action = {
		name = ACTION_NAME_OK
		command = { type = clrflag which = divided_berlin }
	}
}
